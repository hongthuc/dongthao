    // Khởi tạo Slider 2 Banner tự trượt chuyên nghiệp
    function initBannerSlider() {
      const slides = document.querySelectorAll('.banner-slide');
      const dots = document.querySelectorAll('.banner-dot');
      let currentSlide = 0;

      function showSlide(index) {
        currentSlide = (index + slides.length) % slides.length;
        slides.forEach((slide, idx) => {
          if (idx === currentSlide) {
            slide.classList.remove('opacity-0', 'z-0');
            slide.classList.add('opacity-100', 'z-10');
          } else {
            slide.classList.remove('opacity-100', 'z-10');
            slide.classList.add('opacity-0', 'z-0');
          }
        });
        dots.forEach((dot, idx) => {
          dot.className = idx === currentSlide
            ? 'banner-dot w-2.5 h-2.5 rounded-full bg-amber-400 transition'
            : 'banner-dot w-2.5 h-2.5 rounded-full bg-white/50 transition';
        });
      }

      let slideInterval = setInterval(() => {
        showSlide(currentSlide + 1);
      }, 4500);

      function resetTimer() {
        clearInterval(slideInterval);
        slideInterval = setInterval(() => {
          showSlide(currentSlide + 1);
        }, 4500);
      }

      document.getElementById('prevBannerBtn').addEventListener('click', () => {
        showSlide(currentSlide - 1);
        resetTimer();
      });

      document.getElementById('nextBannerBtn').addEventListener('click', () => {
        showSlide(currentSlide + 1);
        resetTimer();
      });

      dots.forEach((dot) => {
        dot.addEventListener('click', () => {
          const idx = parseInt(dot.getAttribute('data-slide'), 10);
          showSlide(idx);
          resetTimer();
        });
      });
    }

    // Tìm kiếm lọc nhanh sản phẩm trên trang
    function filterProductsByQuery(query) {
      const keyword = query.trim().toLowerCase();
      const cards = document.querySelectorAll('.product-card');
      const statusBanner = document.getElementById('searchStatusBanner');
      const statusText = document.getElementById('searchStatusText');

      if (!keyword) {
        cards.forEach((card) => card.classList.remove('hidden'));
        statusBanner.classList.add('hidden');
        return;
      }

      let matchCount = 0;
      cards.forEach((card) => {
        const name = card.getAttribute('data-name') || '';
        if (name.includes(keyword)) {
          card.classList.remove('hidden');
          matchCount++;
        } else {
          card.classList.add('hidden');
        }
      });

      statusBanner.classList.remove('hidden');
      statusText.textContent = `Tìm thấy ${matchCount} sản phẩm phù hợp với từ khóa "${query}"`;
    }
